     1→// Al Gurg Furniture Group - Production and Operations Management System
     2→// Complete schema covering Enquiry, Estimation, Quotation, Approval, Order,
     3→// BOQ, Procurement (MRQ, PR, LPO), QC, Store (GRN), Production (Joinery and Metal
     4→// with QC gates), Drawing, Delivery, Installation and Feedback.
     5→//
     6→// DATABASE PROVIDER
     7→// This schema works with SQLite (development default) or PostgreSQL/MySQL (production).
     8→// To switch to PostgreSQL change provider below to "postgresql" and set
     9→// DATABASE_URL to your production connection string, then run bun run db:push.
    10→
    11→generator client {
    12→  provider = "prisma-client-js"
    13→}
    14→
    15→datasource db {
    16→  // Change to "postgresql" or "mysql" for production deployments.
    17→  provider = "sqlite"
    18→  url      = env("DATABASE_URL")
    19→}
